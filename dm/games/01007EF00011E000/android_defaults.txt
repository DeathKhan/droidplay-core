# Switches this game gets on Android (build_android_from_nsp.py puts them in the game's defaults.txt; env.txt and the settings screen still win)
# DM_VK_MID_DRAWS: a chunk goes to the GPU after 512 draws in the middle of a pass instead of 64 (gfx_vk.c mid_poll). About 3,600 draws a frame in long passes: on a
# OnePlus Pad 3 the replay took 30.6 ms of CPU a frame instead of 32.2 (fewer submissions) and the GPU 18.7 ms instead of 21.6 (fewer passes begun again with their targets
# loaded), the Shrine of Resurrection save, 2026-11-10
DM_VK_MID_DRAWS=512
# DM_VK_RECORDER: the replay's vkCmd* calls and submissions on a thread of their own (gfx_vk_rec.inc). DM_FPS_TARGET=30: the frame step holds 30 (2 vsyncs) as its floor, the
# Switch's own rate. Together on a OnePlus Pad 3 (the Shrine of Resurrection save, 2026-11-10): 30.0 fps, frame p95 38.2-38.9 ms, the replay 17.2-17.5 ms of CPU a frame;
# with the 30 target alone 29.2-29.6 fps (frame p95 41-44 ms); with the recorder alone at the 60 target the step down from 20 fps came in one run of five (the clocks stay low at hold 3)
DM_VK_RECORDER=1
DM_FPS_TARGET=30
